from pathlib import Path
import re
names = ['AGENTS.md', 'CLAUDE.md', 'README.md',
         'docs/YENI-KURULUM-VE-CALISMA.md']
missing = [str(p.parent / link)
           for name in names for p in [Path(name)]
           for link in re.findall(r'\[[^]]*\]\(([^)]+)\)', p.read_text(encoding='utf-8'))
           if not link.startswith(('http:', 'https:', '#'))
           and not (p.parent / link.split('#')[0]).exists()]
assert not missing, missing
a = Path('AGENTS.md').read_text(encoding='utf-8').split('## Yeni oturumda', 1)[1]
b = Path('CLAUDE.md').read_text(encoding='utf-8').split('## Yeni oturumda', 1)[1]
